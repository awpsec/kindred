cmake_minimum_required(VERSION 3.19)
project(kindred_dictation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
option(KINDRED_CPU_AVX2 "Build the separately selected AVX2 CPU worker" OFF)
set(GGML_AVX ${KINDRED_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_AVX2 ${KINDRED_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_FMA ${KINDRED_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_F16C ${KINDRED_CPU_AVX2} CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
add_subdirectory(${WHISPER_SOURCE} whisper)
add_executable(kindred-whisper worker.cpp)
target_link_libraries(kindred-whisper PRIVATE whisper)
if(MINGW)
  target_link_options(kindred-whisper PRIVATE -static -static-libgcc -static-libstdc++ -municode)
endif()
